# git-mcp Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a read-only git operations MCP server in Go that gives AI agents controlled, audited access to source code.

**Architecture:** Single Go binary with four layers — transport (stdio + Streamable HTTP), auth (JWT with issuer-bound validation), tool handlers (six MCP tools shelling out to `git` CLI), and a repo cache manager (bare mirrors on local disk with LRU eviction). Stateless per request.

**Tech Stack:** Go 1.23+, `modelcontextprotocol/go-sdk` (v1.7+), `git` CLI (2.38+)

**Spec:** `docs/superpowers/specs/2026-10-11-git-mcp-design.md`

## Global Constraints

- Go 1.23+ with modules
- `modelcontextprotocol/go-sdk` for MCP server — no other MCP libraries
- Shell out to `git` CLI for all git operations — no go-git, no libgit2
- All git commands use `--end-of-options` and `--` to separate options from arguments
- Every tool response includes `resolved_sha`
- Structured JSON audit logging (one line per tool call)
- No `.git` directory access, no absolute paths, no symlink escape
- Tests use Go's `testing` package + `testify/assert`

```go
package config

import (
    "fmt"
    "os"
    "path/filepath"
    "strings"
    "time"

    "gopkg.in/yaml.v3"
)

type Config struct {
    Server   ServerConfig              `yaml:"server"`
    Upstream map[string]UpstreamConfig `yaml:"upstream"`
    Auth     AuthConfig                `yaml:"auth"`
    Services map[string]*ServiceConfig `yaml:"services"`
}

type ServerConfig struct {
    Listen       string        `yaml:"listen"`
    CacheDir     string        `yaml:"cache_dir"`
    MaxCacheSize string        `yaml:"max_cache_size"`
    MaxFileSize  string        `yaml:"max_file_size"`
    MaxMatches   int           `yaml:"max_matches"`
    MaxDiffSize  string        `yaml:"max_diff_size"`
    GitTimeout   time.Duration `yaml:"git_timeout"`
}

type UpstreamConfig struct {
    Token string `yaml:"token"`
}

type AuthConfig struct {
    Audience string         `yaml:"audience"`
    Issuers  []IssuerConfig `yaml:"issuers"`
}

type IssuerConfig struct {
    Name    string `yaml:"name"`
    Iss     string `yaml:"iss"`
    JWKSUri string `yaml:"jwks_uri"`
}

type ServiceConfig struct {
    Issuer  string   `yaml:"issuer"`
    Subject string   `yaml:"subject"`
    Repos   []string `yaml:"repos"`
}

func (s *ServiceConfig) AllowsRepo(repo string) bool {
    for _, pattern := range s.Repos {
        if pattern == repo {
            return true
        }
        if strings.HasSuffix(pattern, "/*") {
            prefix := strings.TrimSuffix(pattern, "/*")
            dir := filepath.Dir(repo)
            if dir == prefix {
                return true
            }
        }
    }
    return false
}

func Load(path string) (*Config, error) {
    data, err := os.ReadFile(path)
    if err != nil {
        return nil, fmt.Errorf("read config: %w", err)
    }

    expanded := os.ExpandEnv(string(data))

    var cfg Config
    if err := yaml.Unmarshal([]byte(expanded), &cfg); err != nil {
        return nil, fmt.Errorf("parse config: %w", err)
    }

    if cfg.Server.MaxMatches == 0 {
        cfg.Server.MaxMatches = 50
    }
    if cfg.Server.GitTimeout == 0 {
        cfg.Server.GitTimeout = 30 * time.Second
    }

    return &cfg, nil
}
```

```go
package validate

import (
    "fmt"
    "regexp"
    "strings"
)

var (
    hexSHARegex  = regexp.MustCompile(`^[0-9a-f]{7,40}$`)
    refNameRegex = regexp.MustCompile(`^[A-Za-z0-9._][A-Za-z0-9._/-]*$`)
)

func Ref(ref string) error {
    if ref == "" {
        return fmt.Errorf("ref is empty")
    }
    if hexSHARegex.MatchString(ref) {
        return nil
    }
    if !refNameRegex.MatchString(ref) {
        return fmt.Errorf("ref contains invalid characters: %q", ref)
    }
    if strings.Contains(ref, "..") {
        return fmt.Errorf("ref contains '..': %q", ref)
    }
    if strings.Contains(ref, "@{") {
        return fmt.Errorf("ref contains '@{': %q", ref)
    }
    if strings.Contains(ref, "^{") {
        return fmt.Errorf("ref contains '^{': %q", ref)
    }
    if strings.HasPrefix(ref, "-") {
        return fmt.Errorf("ref starts with '-': %q", ref)
    }
    if strings.HasSuffix(ref, ".lock") {
        return fmt.Errorf("ref ends with '.lock': %q", ref)
    }
    return nil
}

func Repo(repo string) error {
    if repo == "" {
        return fmt.Errorf("repo is empty")
    }
    parts := strings.Split(repo, "/")
    if len(parts) < 3 {
        return fmt.Errorf("repo must be host/owner/name: %q", repo)
    }
    for _, part := range parts {
        if part == "" || part == "." || part == ".." {
            return fmt.Errorf("repo contains invalid segment: %q", repo)
        }
        if strings.HasPrefix(part, "-") {
            return fmt.Errorf("repo segment starts with '-': %q", repo)
        }
        if !regexp.MustCompile(`^[A-Za-z0-9._-]+$`).MatchString(part) {
            return fmt.Errorf("repo segment contains invalid characters: %q", repo)
        }
    }
    return nil
}

func Path(path string) error {
    if path == "" {
        return nil
    }
    if strings.HasPrefix(path, "/") {
        return fmt.Errorf("path is absolute: %q", path)
    }
    parts := strings.Split(path, "/")
    for _, part := range parts {
        if part == ".." {
            return fmt.Errorf("path contains '..': %q", path)
        }
        if part == ".git" {
            return fmt.Errorf("path accesses .git: %q", path)
        }
    }
    return nil
}

func Pattern(pattern string) error {
    if pattern == "" {
        return fmt.Errorf("pattern is empty")
    }
    return nil
}
```

```go
package git

import (
    "bytes"
    "context"
    "fmt"
    "os"
    "os/exec"
    "strings"
    "time"
)

type Executor struct {
    timeout time.Duration
}

func NewExecutor(timeout time.Duration) *Executor {
    return &Executor{timeout: timeout}
}

func (e *Executor) Run(ctx context.Context, repoPath string, args ...string) (string, error) {
    ctx, cancel := context.WithTimeout(ctx, e.timeout)
    defer cancel()

    cmd := exec.CommandContext(ctx, "git", args...)
    cmd.Dir = repoPath
    cmd.Env = sanitizeEnv()

    var stdout, stderr bytes.Buffer
    cmd.Stdout = &stdout
    cmd.Stderr = &stderr

    if err := cmd.Run(); err != nil {
        if ctx.Err() == context.DeadlineExceeded {
            return "", fmt.Errorf("git command timed out after %s", e.timeout)
        }
        return "", fmt.Errorf("git %s: %s: %w", strings.Join(args, " "), stderr.String(), err)
    }

    return strings.TrimRight(stdout.String(), "\n"), nil
}

func sanitizeEnv() []string {
    dangerous := map[string]bool{
        "GIT_DIR":           true,
        "GIT_WORK_TREE":     true,
        "GIT_INDEX_FILE":    true,
        "GIT_OBJECT_DIRECTORY": true,
        "GIT_ALTERNATE_OBJECT_DIRECTORIES": true,
        "GIT_ASKPASS":       true,
        "GIT_SSH":           true,
        "GIT_SSH_COMMAND":   true,
        "GIT_CONFIG_GLOBAL": true,
        "GIT_CONFIG_SYSTEM": true,
    }

    var env []string
    for _, e := range os.Environ() {
        key := strings.SplitN(e, "=", 2)[0]
        if !dangerous[key] {
            env = append(env, e)
        }
    }
    env = append(env, "GIT_TERMINAL_PROMPT=0")
    return env
}
```

```go
type UpstreamResolver func(repo string) (url string, token string, err error)

type Manager struct {
    cacheDir string
    executor *git.Executor
    resolve  UpstreamResolver
    mu       sync.Map // per-repo mutexes
}

func NewManager(cacheDir string, executor *git.Executor, resolve UpstreamResolver) *Manager {
    return &Manager{
        cacheDir: cacheDir,
        executor: executor,
        resolve:  resolve,
    }
}

func (m *Manager) repoMutex(repo string) *sync.Mutex {
    v, _ := m.mu.LoadOrStore(repo, &sync.Mutex{})
    return v.(*sync.Mutex)
}

func (m *Manager) RepoPath(repo string) string {
    return filepath.Join(m.cacheDir, repo+".git")
}

func (m *Manager) Ensure(ctx context.Context, repo, ref string, refresh bool) (string, error) {
    repoPath := m.RepoPath(repo)

    mu := m.repoMutex(repo)
    mu.Lock()
    defer mu.Unlock()

    exists := m.repoExists(repoPath)

    if !exists {
        if err := m.cloneRepo(ctx, repo, repoPath); err != nil {
            return "", fmt.Errorf("clone %s: %w", repo, err)
        }
    }

    if refresh && exists {
        if err := m.fetchRepo(ctx, repo, repoPath); err != nil {
            return "", fmt.Errorf("fetch %s: %w", repo, err)
        }
    }

    sha, err := m.resolveRef(ctx, repoPath, ref)
    if err != nil && exists {
        if fetchErr := m.fetchRepo(ctx, repo, repoPath); fetchErr != nil {
            return "", fmt.Errorf("fetch %s: %w", repo, fetchErr)
        }
        sha, err = m.resolveRef(ctx, repoPath, ref)
    }

    if err != nil {
        return "", fmt.Errorf("resolve %s in %s: %w", ref, repo, err)
    }

    return sha, nil
}

func (m *Manager) repoExists(repoPath string) bool {
    _, err := os.Stat(filepath.Join(repoPath, "HEAD"))
    return err == nil
}

func (m *Manager) resolveRef(ctx context.Context, repoPath, ref string) (string, error) {
    out, err := m.executor.Run(ctx, repoPath, "rev-parse", "--verify", "--end-of-options", ref+"^{commit}")
    if err != nil {
        return "", err
    }
    return strings.TrimSpace(out), nil
}

func (m *Manager) cloneRepo(ctx context.Context, repo, repoPath string) error {
    url, token, err := m.resolve(repo)
    if err != nil {
        return err
    }

    if err := os.MkdirAll(filepath.Dir(repoPath), 0o755); err != nil {
        return err
    }

    args := []string{"clone", "--bare"}
    if token != "" {
        args = append(args, "-c", fmt.Sprintf("http.extraHeader=Authorization: Bearer %s", token))
    }
    args = append(args, url, repoPath)

    _, err = m.executor.Run(ctx, ".", args...)
    return err
}

func (m *Manager) fetchRepo(ctx context.Context, repo, repoPath string) error {
    _, token, err := m.resolve(repo)
    if err != nil {
        return err
    }

    args := []string{"fetch", "--prune"}
    if token != "" {
        args = append(args, "-c", fmt.Sprintf("http.extraHeader=Authorization: Bearer %s", token))
    }
    args = append(args, "origin", "+refs/heads/*:refs/heads/*", "+refs/tags/*:refs/tags/*")

    _, err = m.executor.Run(ctx, repoPath, args...)
    return err
}
```

```go
    Path       string `json:"path,omitempty"`

type SearchMatch struct {
    Path       string `json:"path"`
    LineNumber int    `json:"line_number"`
    LineText   string `json:"line_text"`
}

type SearchCodeOutput struct {
    SHA       string        `json:"sha"`
    Matches   []SearchMatch `json:"matches"`
    Truncated bool          `json:"truncated"`
}

type SearchCode struct {
    cache      *cache.Manager
    executor   *git.Executor
    maxMatches int

func NewSearchCode(cache *cache.Manager, executor *git.Executor, maxMatches int) *SearchCode {
    return &SearchCode{cache: cache, executor: executor, maxMatches: maxMatches}
}

func (h *SearchCode) Execute(ctx context.Context, input SearchCodeInput) (*SearchCodeOutput, error) {
    if err := validate.Repo(input.Repo); err != nil {
        return nil, err
    }
    if err := validate.Ref(input.Ref); err != nil {
        return nil, err
    }
    if err := validate.Pattern(input.Pattern); err != nil {
        return nil, err
    }
    if input.Path != "" {
        if err := validate.Path(input.Path); err != nil {
            return nil, err
        }
    }

    sha, err := h.cache.Ensure(ctx, input.Repo, input.Ref, false)
    if err != nil {
        return nil, err
    }

    limit := h.maxMatches
    if input.MaxMatches > 0 && input.MaxMatches < limit {
        limit = input.MaxMatches
    }

    repoPath := h.cache.RepoPath(input.Repo)
    args := []string{"grep", "-n", "-E", fmt.Sprintf("-m%d", limit+1), "-e", input.Pattern, sha}
    if input.Path != "" {
        args = append(args, "--", input.Path)
    }

    out, err := h.executor.Run(ctx, repoPath, args...)
    if err != nil {
        if strings.Contains(err.Error(), "exit status 1") {
            return &SearchCodeOutput{SHA: sha, Matches: []SearchMatch{}}, nil
        }
        return nil, err
    }

    lines := strings.Split(out, "\n")
    truncated := len(lines) > limit
    if truncated {
        lines = lines[:limit]
    }

    matches := make([]SearchMatch, 0, len(lines))
    for _, line := range lines {
        if line == "" {
            continue
        }
        // Format: <sha>:<path>:<line_number>:<text>
        // Strip the sha: prefix
        after := strings.TrimPrefix(line, sha+":")
        parts := strings.SplitN(after, ":", 3)
        if len(parts) < 3 {
            continue
        }
        lineNum, _ := strconv.Atoi(parts[1])
        matches = append(matches, SearchMatch{
            Path:       parts[0],
            LineNumber: lineNum,
            LineText:   parts[2],
        })
    }

    return &SearchCodeOutput{SHA: sha, Matches: matches, Truncated: truncated}, nil
}
```

```go
    "context"
    "fmt"
    "strings"

    "github.com/rophy/git-mcp/internal/cache"

type ReadFileInput struct {
    Repo      string `json:"repo"`
    Ref       string `json:"ref"`
    Path      string `json:"path"`
    StartLine *int   `json:"start_line,omitempty"`
    EndLine   *int   `json:"end_line,omitempty"`
}

type ReadFileOutput struct {
    SHA     string `json:"sha"`
    Content string `json:"content"`
}

type ReadFile struct {
    cache       *cache.Manager
    executor    *git.Executor
    maxFileSize int
}

func NewReadFile(cache *cache.Manager, executor *git.Executor, maxFileSize int) *ReadFile {
    return &ReadFile{cache: cache, executor: executor, maxFileSize: maxFileSize}
}

func (h *ReadFile) Execute(ctx context.Context, input ReadFileInput) (*ReadFileOutput, error) {
    if err := validate.Repo(input.Repo); err != nil {
        return nil, err
    }
    if err := validate.Ref(input.Ref); err != nil {
        return nil, err
    }
    if err := validate.Path(input.Path); err != nil {
        return nil, err
    }
    if input.Path == "" {
        return nil, fmt.Errorf("path is required")
    }

    sha, err := h.cache.Ensure(ctx, input.Repo, input.Ref, false)
    if err != nil {
        return nil, err
    }

    repoPath := h.cache.RepoPath(input.Repo)
    content, err := h.executor.Run(ctx, repoPath, "cat-file", "blob", fmt.Sprintf("%s:%s", sha, input.Path))
    if err != nil {
        return nil, err
    }

    if len(content) > h.maxFileSize {
        return nil, fmt.Errorf("file exceeds max size (%d bytes)", h.maxFileSize)
    }

    lines := strings.Split(content, "\n")
    start := 1
    end := len(lines)
    if input.StartLine != nil {
        start = *input.StartLine
    }
    if input.EndLine != nil {
        end = *input.EndLine
    }
    if start < 1 {
        start = 1
    }
    if end > len(lines) {
        end = len(lines)
    }

    var numbered strings.Builder
    for i := start; i <= end; i++ {
        if i-1 < len(lines) {
            fmt.Fprintf(&numbered, "%d\t%s\n", i, lines[i-1])
        }
    }

    return &ReadFileOutput{SHA: sha, Content: numbered.String()}, nil
}
```

```go
    if err := validate.Repo(input.Repo); err != nil {
        return nil, err
    }
    if err := validate.Ref(input.Ref); err != nil {
        return nil, err
    }
    if input.Path != "" {
        if err := validate.Path(input.Path); err != nil {
            return nil, err
        }
    }

    sha, err := h.cache.Ensure(ctx, input.Repo, input.Ref, false)
    if err != nil {
        return nil, err
    }

    repoPath := h.cache.RepoPath(input.Repo)
    args := []string{"ls-tree"}
    if input.Recursive {
        args = append(args, "-r")
    }
    target := sha
    if input.Path != "" {
        target = sha + ":" + input.Path
    }
    args = append(args, "--end-of-options", target)

    out, err := h.executor.Run(ctx, repoPath, args...)
    if err != nil {
        return nil, err
    }

    var entries []FileEntry
    for _, line := range strings.Split(out, "\n") {
        if line == "" {
            continue
        }
        // Format: <mode> <type> <hash>\t<name>
        tabIdx := strings.IndexByte(line, '\t')
        if tabIdx < 0 {
            continue
        }
        meta := line[:tabIdx]
        name := line[tabIdx+1:]
        parts := strings.Fields(meta)
        if len(parts) < 3 {
            continue
        }
        entryType := "file"
        if parts[1] == "tree" {
            entryType = "directory"
        }
        entries = append(entries, FileEntry{Name: name, Type: entryType})
    }

    return &ListFilesOutput{SHA: sha, Entries: entries}, nil
}
```

```go
    metaOut, err := h.executor.Run(ctx, repoPath, "show", "--no-patch", fmt.Sprintf("--format=%s", format), "--end-of-options", input.Commit)
    if err != nil {
        return nil, err
    }

    fields := strings.SplitN(metaOut, "\x00", 4)
    if len(fields) < 4 {
        return nil, fmt.Errorf("unexpected show output format")
    }

    // Get diff
    diffArgs := []string{"show", "--first-parent", "--format=", "--end-of-options", input.Commit}
    if input.Path != "" {
        diffArgs = append(diffArgs, "--", input.Path)
    }

    diff, err := h.executor.Run(ctx, repoPath, diffArgs...)
    if err != nil {
        return nil, err
    }

    truncated := false
    if len(diff) > h.maxDiffSize {
        diff = diff[:h.maxDiffSize]
        truncated = true
    }

    return &GitShowOutput{
        SHA:       sha,
        Author:    fields[1],
        Date:      fields[2],
        Message:   strings.TrimSpace(fields[3]),
        Diff:      diff,
        Truncated: truncated,
    }, nil
}
```

```go
func RunHTTP(ctx context.Context, srv *mcp.Server, listen string, logger *slog.Logger) error {
    handler := mcp.StreamableHTTPHandler(func() *mcp.Server { return srv }, nil)

    mux := http.NewServeMux()
    mux.Handle("/mcp", handler)
    mux.HandleFunc("/health", func(w http.ResponseWriter, r *http.Request) {
        w.Header().Set("Content-Type", "application/json")
        fmt.Fprintf(w, `{"status":"ok"}`)
    })

    httpSrv := &http.Server{Addr: listen, Handler: mux}
    logger.Info("starting HTTP server", "listen", listen)
    return httpSrv.ListenAndServe()
}

func parseSize(s string, defaultVal int) int {
    if s == "" {
        return defaultVal
    }
    var val int
    var unit string
    fmt.Sscanf(s, "%d%s", &val, &unit)
    switch unit {
    case "KB":
        return val * 1024
    case "MB":
        return val * 1024 * 1024
    case "GB":
        return val * 1024 * 1024 * 1024
    }
    return val
}
```

```go
func main() {
    if len(os.Args) < 2 || os.Args[1] != "serve" {
        fmt.Fprintf(os.Stderr, "Usage: git-mcp serve --config <path> [--stdio]\n")
        os.Exit(1)
    }

    var configPath string
    var useStdio bool
    for i, arg := range os.Args {
        if arg == "--config" && i+1 < len(os.Args) {
            configPath = os.Args[i+1]
        }
        if arg == "--stdio" {
            useStdio = true
        }
    }

    if configPath == "" {
        fmt.Fprintf(os.Stderr, "Error: --config is required\n")
        os.Exit(1)
    }

    cfg, err := config.Load(configPath)
    if err != nil {
        fmt.Fprintf(os.Stderr, "Error: %v\n", err)
        os.Exit(1)
    }

    logger := slog.New(slog.NewJSONHandler(os.Stderr, nil))

    executor := git.NewExecutor(cfg.Server.GitTimeout)

    upstreamResolver := func(repo string) (string, string, error) {
        host := strings.SplitN(repo, "/", 2)[0]
        upstream, ok := cfg.Upstream[host]
        if !ok {
            return "", "", fmt.Errorf("no upstream configured for host %q", host)
        }
        return fmt.Sprintf("https://%s", repo), upstream.Token, nil
    }

    cacheMgr := cache.NewManager(cfg.Server.CacheDir, executor, upstreamResolver)
    srv := server.New(cfg, cacheMgr, executor, logger)

    ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt)
    defer stop()

    if useStdio {
        logger.Info("starting in stdio mode")
        if err := server.RunStdio(ctx, srv); err != nil {
            logger.Error("stdio server error", "err", err)
            os.Exit(1)
        }
    } else {
        listen := cfg.Server.Listen
        if listen == "" {
            listen = "0.0.0.0:8080"
        }
        if err := server.RunHTTP(ctx, srv, listen, logger); err != nil {
            logger.Error("http server error", "err", err)
            os.Exit(1)
        }
    }
}
```

```go
package audit

import (
	"log/slog"
	"time"
)

type Entry struct {
	Service     string
	RequestID   string
	Tool        string
	Repo        string
	Ref         string
	ResolvedSHA string
	Path        string
	Duration    time.Duration
	Error       string
}

type Logger struct {
	log *slog.Logger
}

func NewLogger(log *slog.Logger) *Logger {
	return &Logger{log: log}
}

func (l *Logger) Log(e Entry) {
	attrs := []slog.Attr{
		slog.String("service", e.Service),
		slog.String("tool", e.Tool),
		slog.String("repo", e.Repo),
	}

	if e.RequestID != "" {
		attrs = append(attrs, slog.String("request_id", e.RequestID))
	}
	if e.Ref != "" {
		attrs = append(attrs, slog.String("ref", e.Ref))
	}
	if e.ResolvedSHA != "" {
		attrs = append(attrs, slog.String("resolved_sha", e.ResolvedSHA))
	}
	if e.Path != "" {
		attrs = append(attrs, slog.String("path", e.Path))
	}

	attrs = append(attrs, slog.Float64("duration_ms", float64(e.Duration.Milliseconds())))

	if e.Error != "" {
		attrs = append(attrs, slog.String("error", e.Error))
	}

	args := make([]any, len(attrs))
	for i, a := range attrs {
		args[i] = a
	}
	l.log.Info("tool_call", args...)
}
```

```go
package audit

import (
	"context"
	"net/http"
	"strings"
)

type contextKey string

const (
	requestIDKey contextKey = "request_id"
	serviceKey   contextKey = "service"
)

func RequestIDFromContext(ctx context.Context) string {
	v, _ := ctx.Value(requestIDKey).(string)
	return v
}

func ServiceFromContext(ctx context.Context) string {
	v, _ := ctx.Value(serviceKey).(string)
	return v
}

func ContextWithService(ctx context.Context, service string) context.Context {
	return context.WithValue(ctx, serviceKey, service)
}

func Middleware(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		requestID := r.Header.Get("X-Request-Id")
		if requestID == "" {
			tp := r.Header.Get("Traceparent")
			if tp != "" {
				parts := strings.Split(tp, "-")
				if len(parts) >= 3 {
					requestID = parts[1] + "-" + parts[2]
				}
			}
		}

		ctx := r.Context()
		if requestID != "" {
			ctx = context.WithValue(ctx, requestIDKey, requestID)
		}

		next.ServeHTTP(w, r.WithContext(ctx))
	})
}
```

```go
package auth

import (
	"context"
	"fmt"
	"strings"
	"sync"
	"time"

	gojwt "github.com/golang-jwt/jwt/v5"
	"github.com/lestrrat-go/jwx/v2/jwk"
	"github.com/rophy/git-mcp/internal/config"
)

type Identity struct {
	Service string
	Issuer  string
	Subject string
}

type issuerEntry struct {
	name    string
	iss     string
	jwksURI string
	cache   jwk.Cache
}

type Authenticator struct {
	audience string
	issuers  map[string]*issuerEntry // keyed by iss claim value
	services map[string]config.ServiceConfig
	mu       sync.RWMutex
}

func NewAuthenticator(cfg *config.Config) (*Authenticator, error) {
	issuers := make(map[string]*issuerEntry, len(cfg.Auth.Issuers))

	for _, ic := range cfg.Auth.Issuers {
		c := jwk.NewCache(context.Background())
		err := c.Register(ic.JWKSURI, jwk.WithMinRefreshInterval(5*time.Minute))
		if err != nil {
			return nil, fmt.Errorf("register JWKS for issuer %q: %w", ic.Name, err)
		}

		issuers[ic.Iss] = &issuerEntry{
			name:    ic.Name,
			iss:     ic.Iss,
			jwksURI: ic.JWKSURI,
			cache:   *c,
		}
	}

	return &Authenticator{
		audience: cfg.Auth.Audience,
		issuers:  issuers,
		services: cfg.Services,
	}, nil
}

func (a *Authenticator) Authenticate(tokenString string) (*Identity, error) {
	parser := gojwt.NewParser(gojwt.WithoutClaimsValidation())
	unverified, _, err := parser.ParseUnverified(tokenString, gojwt.MapClaims{})
	if err != nil {
		return nil, fmt.Errorf("parse token: %w", err)
	}

	claims, ok := unverified.Claims.(gojwt.MapClaims)
	if !ok {
		return nil, fmt.Errorf("unexpected claims type")
	}

	iss, _ := claims["iss"].(string)
	if iss == "" {
		return nil, fmt.Errorf("missing iss claim")
	}

	issEntry, ok := a.issuers[iss]
	if !ok {
		return nil, fmt.Errorf("unknown issuer: %s", iss)
	}

	keySet, err := issEntry.cache.Get(context.Background(), issEntry.jwksURI)
	if err != nil {
		return nil, fmt.Errorf("fetch JWKS for %s: %w", issEntry.name, err)
	}

	verified, err := gojwt.Parse(tokenString, func(token *gojwt.Token) (any, error) {
		kid, _ := token.Header["kid"].(string)
		if kid == "" {
			return nil, fmt.Errorf("missing kid header")
		}

		key, found := keySet.LookupKeyID(kid)
		if !found {
			return nil, fmt.Errorf("key %s not found in JWKS", kid)
		}

		var rawKey any
		if err := key.Raw(&rawKey); err != nil {
			return nil, fmt.Errorf("extract key: %w", err)
		}
		return rawKey, nil
	}, gojwt.WithAudience(a.audience))

	if err != nil {
		if strings.Contains(err.Error(), "token is expired") {
			return nil, fmt.Errorf("token expired: %w", err)
		}
		return nil, fmt.Errorf("validate token: %w", err)
	}

	verifiedClaims, ok := verified.Claims.(gojwt.MapClaims)
	if !ok {
		return nil, fmt.Errorf("unexpected claims type after verification")
	}

	sub, _ := verifiedClaims["sub"].(string)
	if sub == "" {
		return nil, fmt.Errorf("missing sub claim")
	}

	serviceName := a.findService(issEntry.name, sub)
	if serviceName == "" {
		return nil, fmt.Errorf("no service matching issuer=%s sub=%s", issEntry.name, sub)
	}

	return &Identity{
		Service: serviceName,
		Issuer:  issEntry.name,
		Subject: sub,
	}, nil
}

func (a *Authenticator) findService(issuerName, subject string) string {
	for name, svc := range a.services {
		if svc.Issuer == issuerName && svc.Subject == subject {
			return name
		}
	}
	return ""
}

func (a *Authenticator) CheckRepoAccess(service, repo string) bool {
	svc, ok := a.services[service]
	if !ok {
		return false
	}
	for _, pattern := range svc.Repos {
		if matchRepo(pattern, repo) {
			return true
		}
	}
	return false
}

func matchRepo(pattern, repo string) bool {
	if pattern == repo {
		return true
	}
	if strings.HasSuffix(pattern, "/*") {
		prefix := strings.TrimSuffix(pattern, "/*")
		return strings.HasPrefix(repo, prefix+"/")
	}
	return false
}
```

Create `internal/auth/middleware.go`:
```go
package auth

import (
	"net/http"
	"strings"

	"github.com/rophy/git-mcp/internal/audit"
)

func HTTPMiddleware(authenticator *Authenticator) func(http.Handler) http.Handler {
	return func(next http.Handler) http.Handler {
		return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
			if r.URL.Path == "/health" {
				next.ServeHTTP(w, r)
				return
			}

			authHeader := r.Header.Get("Authorization")
			if authHeader == "" {
				http.Error(w, `{"error":"missing authorization header"}`, http.StatusUnauthorized)
				return
			}

			token := strings.TrimPrefix(authHeader, "Bearer ")
			if token == authHeader {
				http.Error(w, `{"error":"invalid authorization format"}`, http.StatusUnauthorized)
				return
			}

			identity, err := authenticator.Authenticate(token)
			if err != nil {
				http.Error(w, `{"error":"authentication failed"}`, http.StatusUnauthorized)
				return
			}

			ctx := audit.ContextWithService(r.Context(), identity.Service)
			next.ServeHTTP(w, r.WithContext(ctx))
		})
	}
}
```

```go
func RunHTTP(ctx context.Context, srv *mcp.Server, listen string, authenticator *auth.Authenticator, logger *slog.Logger) error {
	handler := mcp.StreamableHTTPHandler(func() *mcp.Server { return srv }, nil)

	mux := http.NewServeMux()
	mux.Handle("/mcp", handler)
	mux.HandleFunc("/health", func(w http.ResponseWriter, r *http.Request) {
		w.Header().Set("Content-Type", "application/json")
		fmt.Fprintf(w, `{"status":"ok"}`)
	})

	var h http.Handler = mux
	h = audit.Middleware(h)
	if authenticator != nil {
		h = auth.HTTPMiddleware(authenticator)(h)
	}

	httpSrv := &http.Server{Addr: listen, Handler: h}
	logger.Info("starting HTTP server", "listen", listen)
	return httpSrv.ListenAndServe()
}
```
